addpath('/Users/bensonossai/Desktop/MATLAB_TOOLS/github_repo/');%cmocean colormaps

%% find grid domain to extract
g=roms_get_grid('grid_eccofs_3km_08_b7_w3.nc');%this is the eccofs grid
figure(1);
h_plot=g.h;h_plot(g.mask_rho==0)=NaN;%mask land
pcolor(g.lon_rho,g.lat_rho,h_plot);
shading flat
hold on
x1=597;x2=998;y1=89;y2=618;%specify the sub-domain
plot(g.lon_rho(:,x1),g.lat_rho(:,x1),'r','linewidth',2)
plot(g.lon_rho(:,x2),g.lat_rho(:,x2),'r','linewidth',2)
plot(g.lon_rho(y1,:),g.lat_rho(y1,:),'r','linewidth',2)
plot(g.lon_rho(y2,:),g.lat_rho(y2,:),'r','linewidth',2)
cmocean('deep');
grid on
[ny_full,nx_full]=size(g.h);%full grid size
title(sprintf('nx=%d ny=%d',nx_full,ny_full));


%extratct the sub-domain
lon_box=g.lon_rho(y1:y2,x1:x2);
lat_box=g.lat_rho(y1:y2,x1:x2);
h_box=g.h(y1:y2,x1:x2);
mask_box=g.mask_rho(y1:y2,x1:x2);
h_box(mask_box==0)=NaN;%mask land

%plot extracted sub-domain
fig2=figure(2);clf
pcolor(lon_box,lat_box,h_box);
shading flat
hold on
cmocean('deep');%pale shelf -> dark abyss
set(gca,'ColorScale','log');%log scale resolves both shelf and deep-ocean depths
clim([5 max(h_box(:))]);
cb=colorbar;cb.Label.String='Depth (m)';cb.Ticks=[10 20 50 100 200 500 1000 2000 5000];
xlabel('Longitude');ylabel('Latitude');
axis equal
axis tight
grid on
box on
[ny,nx]=size(h_box);%rows=eta (y), columns=xi (x)
title(sprintf('nx=%d ny=%d',nx,ny));
set(gca,'FontSize',15);
